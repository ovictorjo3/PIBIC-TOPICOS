\section{Runge-Kutta - 2ª ordem}

	\begin{frame}{Runge-Kutta - 2ª ordem (RK2)}
		Considere:
		\begin{itemize}
			\item $y' = f(x,y)$;
			\item $x_{k+1}=x_{k}+h$ e;
			\item $y_{k+1}=y_{k}+\dfrac{h}{2}\left[ f(x_{k},y_{k})+f(x_{k}+h,y_{k}+hf(x_{k},y_{k}) \right]$
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Runge-Kutta - 2ª ordem (RK2)}
		Exemplo
		\begin{itemize}
			\item $5y'+y^{2}-24x=0$;
			\item $y(0)=5$;
			\item $h=1$;
			\item $y(2)=?$
		\end{itemize}
	\end{frame}

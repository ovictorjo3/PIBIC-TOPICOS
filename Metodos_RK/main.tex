\section{Runge-Kutta - 2ª ordem}

	\begin{frame}{Runge-Kutta - 2ª ordem (RK2)}
		Considere:
		\begin{itemize}
			\item $y' = f(x,y)$;
			\item $x_{k+1}=x_{k}+h$ e;
			\item $y_{k+1}=y_{k}+\dfrac{h}{2}\left[ f(x_{k},y_{k})+f(x_{k}+h,y_{k}+hf(x_{k},y_{k}) \right]$
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Runge-Kutta - 2ª ordem (RK2)}
		Exemplo
		\begin{itemize}
			\item $5y'+y^{2}-24x=0$;
			\item $y(0)=5$;
			\item $h=1$;
			\item $y(2)=?$
		\end{itemize}

	\end{frame}

\section{Runge-Kutta - 3ª ordem}

	\begin{frame}{Runge-Kutta - 3ª ordem (RK3)}
		\begin{itemize}
			\item $y'=f(x,y)$;
			\item $x_{n+1}=x_ {n}+h$;
			\item $y_{n+1}=y_{n}+\dfrac{2}{9}k_{1}+\dfrac{1}{3}k_{2}+
			\dfrac{4}{9}k_{3}$;
			\item $k_{1}=hf(x_{n},y_{n})$;
			\item $k_{2}=hf(x_{n}+{h}/{2};y_{n}+{k_{1}}/{2})$;
			\item $k_{3}=hf(x_{n}+{3h}/{4};y_{n}+{3k_{2}}/{4})$
		\end{itemize}
	\end{frame}
	
	\begin{frame}{Runge-Kutta - 3ª ordem (RK3)}
		Exemplo
		\begin{itemize}
			\item $y'=x+y$;
			\item $y(0)=10$;
			\item $h=1$;
			\item $y(1)=?$
		\end{itemize}
